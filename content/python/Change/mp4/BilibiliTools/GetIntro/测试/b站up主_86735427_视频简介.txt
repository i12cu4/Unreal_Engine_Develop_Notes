import maya.cmds as cmds
import random

# Get All List
selected_objs = cmds.ls(sl=True)

# Run
count = 1

# All Mesh
for obj in selected_objs:
  # Create blinn 
  new_mat = cmds.shadingNode('blinn', asShader=True, name='ID' + str(count))
  cmds.sets(name='ID' + str(count) + '_SG', renderable=True, noSurfaceShader=True, empty=True)
  cmds.connectAttr(new_mat+'.outColor', 'ID' + str(count) + '_SG.surfaceShader', force=True)
  cmds.sets(obj, edit=True, forceElement='ID' + str(count) + '_SG')

  # Random
  r, g, b = random.uniform(0, 1), random.uniform(0, 1), random.uniform(0, 1)
  cmds.setAttr(new_mat+'.color', r, g, b, type='double3')

  # +1
  count += 1

# All"Blinn"
Blinn_mats = cmds.ls(type="Blinn")

for mat in Blinn_mats:
  # Detact
  if not cmds.listConnections(mat+".outColor"):
    # Del
    cmds.delete(mat)
    print("Deleted unused Blinn material: " + mat)
